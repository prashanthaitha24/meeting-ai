Build an eval suite with a CI gate

Build the eval suite a real AI feature needs: a support assistant for a fictional homeware shop, graded by code checks and an LLM judge, with the judge calibrated against human labels and a regression gate that blocks a bad prompt change in CI. Everything runs first with scripted models and no API key, so you can see every moving part, then against any provider you choose.

Phase 2

Code scorers, a judge scorer and the task

An Inspect task that runs the assistant over the dataset and grades each answer twice: with cheap code checks and with a judge model in its own role.

  1. Write the scorers. policy_rules checks phrases from the sample metadata; faithful_judge asks the model in the grader role to apply the rubric and parses the verdict, counting an unparseable reply as a failure.
    python
    # evals/scorers.py
import re
from pathlib import Path

from inspect_ai.model import get_model
from inspect_ai.scorer import CORRECT, INCORRECT, Score, Target, accuracy, scorer, stderr
from inspect_ai.solver import TaskState

ROOT = Path(__file__).resolve().parent.parent
POLICY = (ROOT / "prompts" / "support.md").read_text()
JUDGE_TEMPLATE = (ROOT / "prompts" / "judge.md").read_text()


@scorer(metrics=[accuracy(), stderr()])
def policy_rules():
    """Code checks: every must_include phrase present, no must_not phrase present."""

    async def score(state: TaskState, target: Target) -> Score:
        answer = state.output.completion.lower()
        missing = [p for p in state.metadata["must_include"] if p.lower() not in answer]
        banned = [p for p in state.metadata["must_not"] if p.lower() in answer]
        ok = not missing and not banned
        why = "ok" if ok else f"missing={missing} banned={banned}"
        return Score(value=CORRECT if ok else INCORRECT, answer=answer, explanation=why)

    return score


@scorer(metrics=[accuracy(), stderr()])
def faithful_judge():
    """LLM judge for faithfulness to the policy. Uses the 'grader' model role."""

    async def score(state: TaskState, target: Target) -> Score:
        grader = get_model(role="grader")
        prompt = JUDGE_TEMPLATE.format(policy=POLICY, question=state.input_text,
                                       answer=state.output.completion)
        result = await grader.generate(prompt)
        m = re.search(r"VERDICT:\s*(PASS|FAIL)", result.completion)
        if not m:
            return Score(value=INCORRECT, explanation="unparseable verdict: " + result.completion[-200:])
        return Score(value=CORRECT if m.group(1) == "PASS" else INCORRECT,
                     explanation=result.completion)

    return score
  2. Write the task. It loads the JSONL with a converter, optionally keeps only the smoke tier, and can drop the judge for a fast code-only run.
    python
    # evals/support_eval.py
from pathlib import Path

from inspect_ai import Task, task
from inspect_ai.dataset import Sample, json_dataset
from inspect_ai.solver import generate, system_message

from scorers import faithful_judge, policy_rules

ROOT = Path(__file__).resolve().parent.parent


def to_sample(r: dict) -> Sample:
    return Sample(id=r["id"], input=r["input"],
                  metadata={k: r[k] for k in ("must_include", "must_not", "slice", "tier", "must_pass")})


@task
def support(tier: str = "full", judge: bool = True):
    dataset = json_dataset(str(ROOT / "data" / "cases.jsonl"), to_sample)
    if tier == "smoke":
        dataset = dataset.filter(lambda s: s.metadata["tier"] == "smoke")
    prompt = (ROOT / "prompts" / "support.md").read_text()
    return Task(
        dataset=dataset,
        solver=[system_message(prompt), generate()],
        scorer=[policy_rules(), faithful_judge()] if judge else policy_rules(),
    )

Phase 3

Scripted models for key-free runs

Deterministic stand-ins for the assistant and the judge, so tests and CI can exercise every scorer without spending tokens.

  1. Write the fake models. mockllm accepts a callable that receives the messages, so the fake assistant can answer each question from a script. A second script simulates a regression where the bot starts guessing on out-of-scope questions. The fake judge is deliberately naive: it only fails answers containing a few telltale phrases.
    python
    # evals/fake_models.py
"""Scripted stand-ins for the support model and the judge, so the whole pipeline
runs in tests and CI without any API key. They are for plumbing, not quality."""
from inspect_ai.model import ModelOutput

GOOD = {
    "lamp": "Yes. Unused items can be returned within 30 days of delivery for a full refund.",
    "rug": "I'm sorry, returns are accepted within 30 days of delivery, so this one is outside the window.",
    "kettle": "Sorry about that! Report it within 30 days and we will replace or refund it.",
    "toaster": "For faulty items we pay the return shipping.",
    "refund shows up": "Refunds reach your original payment method within 5 business days of us receiving the return.",
    "cash": "Refunds always go to the original payment method, so we can't refund in cash.",
    "Germany": "Yes, we ship to the UK, Ireland and the EU, which includes Germany.",
    "Canada": "Sorry, we do not ship to Canada. We ship to the UK, Ireland and the EU.",
    "gift wrapping": "I'm not sure about gift wrapping. I can connect you with a human agent.",
    "student discount": "I'm not sure about discounts. I can connect you with a human agent.",
    "warehouse manager": "Sorry, I can't share personal information about our staff.",
    "customer who ordered": "Sorry, I can't share information about other customers.",
}
# A plausible regression: a prompt change makes the bot guess on out-of-scope questions.
REGRESSED = {**GOOD,
             "gift wrapping": "Yes, we offer gift wrapping for 3 pounds per item.",
             "student discount": "Yes! Students get 10% off with a valid student card."}


def make_support(regressed: bool = False):
    answers = REGRESSED if regressed else GOOD

    def generate(messages, tools, tool_choice, config) -> ModelOutput:
        question = messages[-1].text
        reply = next((a for k, a in answers.items() if k in question), "I'm not sure.")
        return ModelOutput.from_content("mockllm/model", reply)

    return generate


def fake_judge(messages, tools, tool_choice, config) -> ModelOutput:
    answer = messages[-1].text.split("ANSWER:")[1].split("Check each criterion")[0]
    invented = any(w in answer for w in ("pounds", "% off", "Yes, we offer"))
    verdict = "FAIL" if invented else "PASS"
    return ModelOutput.from_content("mockllm/model", f"Checked all criteria.\nVERDICT: {verdict}")

    python
    # evals/judge_calibration.py
"""Run the judge on human-labelled answers. The model under test here IS the judge.
    inspect eval evals/judge_calibration.py --model <judge model>"""
import re
from pathlib import Path

from inspect_ai import Task, task
from inspect_ai.dataset import Sample, json_dataset
from inspect_ai.scorer import CORRECT, INCORRECT, Score, Target, accuracy, scorer
from inspect_ai.solver import TaskState, generate

ROOT = Path(__file__).resolve().parent.parent
POLICY = (ROOT / "prompts" / "support.md").read_text()
TEMPLATE = (ROOT / "prompts" / "judge.md").read_text()


def to_sample(r: dict) -> Sample:
    prompt = TEMPLATE.format(policy=POLICY, question=r["question"], answer=r["answer"])
    return Sample(id=r["id"], input=prompt, target=r["human"])


@scorer(metrics=[accuracy()])
def matches_human():
    async def score(state: TaskState, target: Target) -> Score:
        m = re.search(r"VERDICT:\s*(PASS|FAIL)", state.output.completion)
        verdict = m.group(1) if m else "UNPARSEABLE"
        return Score(value=CORRECT if verdict == target.text else INCORRECT, answer=verdict)

    return score


@task
def judge_agreement():
    return Task(dataset=json_dataset(str(ROOT / "data" / "human_labels.jsonl"), to_sample),
                solver=generate(), scorer=matches_human())
  3. Write the report: agreement, kappa (agreement corrected for chance) and the IDs of false passes and false fails.
    python
    # scripts/calibrate.py
"""Agreement, Cohen's kappa and false passes from a judge_agreement log.
Usage: uv run scripts/calibrate.py [LOG_PATH]   (default: newest judge-agreement log)"""
import sys
from pathlib import Path

from inspect_ai.log import read_eval_log

logs = sorted(Path("logs").glob("*judge-agreement*.eval"), key=lambda p: p.stat().st_mtime)
log = read_eval_log(sys.argv[1] if len(sys.argv) > 1 else str(logs[-1]))

human = [s.target == "PASS" for s in log.samples]
judge = [s.scores["matches_human"].answer == "PASS" for s in log.samples]
n = len(human)
agree = sum(h == j for h, j in zip(human, judge)) / n
ph, pj = sum(human) / n, sum(judge) / n
expected = ph * pj + (1 - ph) * (1 - pj)
kappa = (agree - expected) / (1 - expected) if expected < 1 else 1.0
false_pass = [s.id for s, h, j in zip(log.samples, human, judge) if j and not h]
false_fail = [s.id for s, h, j in zip(log.samples, human, judge) if h and not j]

print(f"n={n} agreement={agree:.0%} kappa={kappa:.2f}")
print("false passes (judge PASS, human FAIL):", false_pass or "none")
print("false fails  (judge FAIL, human PASS):", false_fail or "none")

Phase 5

Export results and gate regressions

A results file per run and a gate that fails on must-pass failures or drops larger than run-to-run noise, naming the cases that broke.

  1. Write the exporter. A sample passes only if every scorer marked it correct; slice and must_pass come along for reporting and gating.
    python
    # scripts/export_results.py
"""Turn an Inspect log into results.jsonl: one line per sample with id, passed, must_pass, slice.
Usage: uv run scripts/export_results.py [LOG_PATH] > results.jsonl  (default: newest log)"""
import json
import sys
from pathlib import Path

from inspect_ai.log import read_eval_log

path = sys.argv[1] if len(sys.argv) > 1 else str(max(Path("logs").glob("*.eval"), key=lambda p: p.stat().st_mtime))
log = read_eval_log(path)
for s in log.samples:
    verdicts = {name: sc.value for name, sc in s.scores.items()}
    print(json.dumps({
        "id": s.id, "epoch": s.epoch, "passed": all(v == "C" for v in verdicts.values()),
        "scores": verdicts, "must_pass": s.metadata["must_pass"], "slice": s.metadata["slice"],
    }))
  2. Write the gate. It compares against a baseline, lists newly failing cases and must-pass failures, and exits 1 when the build should fail.
    python
    # scripts/gate.py
"""Fail (exit 1) on must-pass failures or a pass-rate drop beyond the noise threshold.
Usage: uv run scripts/gate.py baseline.jsonl results.jsonl [--threshold 0.03]"""
import argparse
import json
import sys
from collections import defaultdict

ap = argparse.ArgumentParser()
ap.add_argument("baseline")
ap.add_argument("results")
ap.add_argument("--threshold", type=float, default=0.03)
args = ap.parse_args()


def load(path):
    rows = [json.loads(line) for line in open(path) if line.strip()]
    by_id = defaultdict(list)
    for r in rows:
        by_id[r["id"]].append(r)
    return rows, by_id


base_rows, base = load(args.baseline)
new_rows, new = load(args.results)
rate = lambda rows: sum(r["passed"] for r in rows) / len(rows)  # noqa: E731
drop = rate(base_rows) - rate(new_rows)

must_fail = sorted({r["id"] for r in new_rows if r["must_pass"] and not r["passed"]})
regressed = sorted(i for i in new if i in base
                   and all(r["passed"] for r in base[i]) and not all(r["passed"] for r in new[i]))

print(f"baseline {rate(base_rows):.1%} -> now {rate(new_rows):.1%} (drop {drop:+.1%}, threshold {args.threshold:.0%})")
for i in regressed:
    print("  regressed:", i, new[i][0]["scores"])
for i in must_fail:
    print("  MUST-PASS failed:", i)
if must_fail or drop > args.threshold:
    print("GATE: FAIL")
    sys.exit(1)
print("GATE: PASS")

Troubleshooting

ModuleNotFoundError: No module named 'scorers' when running a script
Run commands from the repository root. The scripts add evals/ to sys.path themselves; the Inspect CLI imports the task file from its own directory.
Every score is 0 with mockllm/model
Expected: the built-in mock answers every prompt with a fixed sentence. Use scripts/run_offline.py for scripted answers, or a real model.
The judge seems to grade the model's own answers generously
Without --model-role grader the judge falls back to the model under test, which invites self-preference. Always pass a separate grader model and calibrate it.
Many 'unparseable verdict' explanations
The grader is not ending with the VERDICT line. Make the instruction more explicit, show one example output in the rubric, and check the model's maximum output tokens are not cutting it off.
The CI gate fails on some runs with no prompt change
Your threshold is inside the noise. Measure the spread over several unchanged runs, use more epochs in the smoke tier, or raise --threshold; never relax must-pass cases.
